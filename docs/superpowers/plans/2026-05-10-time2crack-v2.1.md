# Time2Crack v2.1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build Time2Crack v2.1 — a static site (Eleventy + Nunjucks) with modular vanilla JS (esbuild), serving 126 HTML pages in 9 languages, replacing the v1 on YunoHost while v1 stays live on GitHub Pages.

**Architecture:** Eleventy generates 126 HTML pages via pagination over a `site.langs` array, using per-language `_data/i18n/{lang}.json` files with `throwOnUndefined: true` to catch missing keys at build time. JavaScript is written as ES6 modules in `src/js/`, bundled by esbuild into a single `dist/app.bundled.js`. Core engine modules (`core/`, `attacks/`) are ported from `v1/core/` and `v1/core/rank/` with TDD; UI modules (`ui/`) wire DOM events to the engine.

**Tech Stack:** Eleventy 3.x, Nunjucks, esbuild 0.21, Vitest 1.x, Playwright 1.x, axe-core/playwright 4.x, fake-indexeddb 5.x, husky 9.x, lint-staged 15.x, cssnano 7.x

```js
import { readdirSync } from 'fs';
import { join } from 'path';

function countHtml(dir) {
  let count = 0;
  for (const entry of readdirSync(dir, { withFileTypes: true })) {
    if (entry.isDirectory()) count += countHtml(join(dir, entry.name));
    else if (entry.name.endsWith('.html')) count++;
  }
  return count;
}

const count = countHtml('dist');
const expected = 126;

if (count !== expected) {
  console.error(`FAIL: ${count} fichiers HTML trouvés (attendu ${expected})`);
  process.exit(1);
}
console.log(`✓ Build check: ${count} fichiers HTML générés`);
```

```js
export function formatTime(seconds) {
  if (seconds === 0) return 'instant';
  if (!isFinite(seconds)) return 'centuries';

  const units = [
    [31536000, 'year', 'years'],
    [86400,    'day',  'days'],
    [3600,     'hour', 'hours'],
    [60,       'minute', 'minutes'],
    [1,        'second', 'seconds'],
  ];

  for (const [divisor, singular, plural] of units) {
    const value = Math.floor(seconds / divisor);
    if (value >= 1) return `${value} ${value === 1 ? singular : plural}`;
  }
  return 'instant';
}

export function formatNumber(n) {
  return n.toLocaleString('en-US');
}
```

```js
export const HASH_RATES = {
  md5:    2_027e9,
  sha1:   610e9,
  sha256: 272e9,
  ntlm:   3_462e9,
  bcrypt: 2_208e3,
  argon2: 800,
};

export const ATTACKER_MULTIPLIERS = {
  amateur:      1 / 12,
  experienced:  1,
  professional: 100 / 12,
  nation_state: 10_000 / 12,
};

export function rankToSeconds(rank, algo, profile = 'experienced') {
  const baseRate = HASH_RATES[algo];
  if (!baseRate) throw new Error(`Algorithme inconnu : ${algo}`);
  const multiplier = ATTACKER_MULTIPLIERS[profile] ?? 1;
  return rank / (baseRate * multiplier);
}

export function rankToAllSeconds(rank, profile = 'experienced') {
  return Object.fromEntries(
    Object.keys(HASH_RATES).map(algo => [algo, rankToSeconds(rank, algo, profile)])
  );
}
```

```js
const KEYBOARD_SEQUENCES = [
  'qwertyuiop', 'asdfghjkl', 'zxcvbnm',
  'azertyuiop', 'qsdfghjklm', 'wxcvbn',
  'qwerty', 'azerty', 'qwert', 'azert',
  'asdf', 'zxcv', 'poiuy', 'lkjhg',
  '1qaz', '2wsx', '3edc', 'qazwsx',
];
const ALPHA_SEQUENCES = ['abcdef', 'abcde', 'abcd', 'bcde', 'cdef'];
const NUM_SEQUENCES   = ['123456', '234567', '12345', '23456', '11111', '00000'];

function reverseStr(s) { return [...s].reverse().join(''); }

export function analyzePatterns(password, dictWords = null) {
  const len   = passwordLength(password);
  const cs    = charsetSize(password);
  const lower = password.toLowerCase();

  const kbPat = KEYBOARD_SEQUENCES.some(seq =>
    lower.includes(seq) || lower.includes(reverseStr(seq))
  );
  const seq = ALPHA_SEQUENCES.some(s => lower.includes(s))
           || NUM_SEQUENCES.some(s => lower.includes(s));
  const rep = (new Set([...lower]).size === 1) && len > 2;

  const DATE_PATTERNS = [
    /\b(19|20)\d{2}(0[1-9]|1[0-2])(0[1-9]|[12]\d|3[01])\b/,
    /\b(0[1-9]|[12]\d|3[01])(0[1-9]|1[0-2])(19|20)\d{2}\b/,
    /\b(19|20)\d{2}\b/,
    /\b(0[1-9]|[12]\d|3[01])(0[1-9]|1[0-2])\d{2}\b/,
  ];
  const dateMatch = DATE_PATTERNS.find(p => p.test(password));
  const dt = Boolean(dateMatch);
  const datePattern = dateMatch ? dateMatch.source : null;

  const norm = lower.replace(/[^a-z]/g, '');
  const dictWord = dictWords !== null && (
    dictWords.has(norm) ||
    (norm.length > 2 && dictWords.has(norm.slice(0, -1)))
  );
  const hybridVuln = dictWord && (
    /[^a-zA-Z]/.test(password) ||
    /[A-Z]/.test(password.slice(1))
  );
  const looksPassphrase = !kbPat && !seq && len >= 15 && password.split(' ').length >= 3;

  return {
    pw: password,
    len,
    cs,
    full: cs ** len,
    kbPat,
    seq,
    rep,
    dt,
    dictWord,
    hybridVuln,
    looksPassphrase,
    datePattern,
  };
}
```

```js
import { analyzePatterns }                              from './patterns.js';
import { estimateRank }                                 from '../attacks/index.js';
import { rankToSeconds, rankToAllSeconds, HASH_RATES }  from './time.js';

export const ALGORITHMS = Object.keys(HASH_RATES);

export function calcCrackTime(password, options = {}) {
  const {
    dictWords   = null,
    isHibpHit   = false,
    hibpRank    = null,
    profile     = 'experienced',
    lang        = 'en',
  } = options;

  const context = analyzePatterns(password, dictWords);
  const rank    = estimateRank(password, {
    dictWords, isHibpHit, hibpRank,
    looksPassphrase: context.looksPassphrase,
    hybridVuln: context.hybridVuln,
    lang,
  });

  const times            = rankToAllSeconds(rank.standard,    profile);
  const times_optimistic = rankToAllSeconds(rank.optimistic,  profile);
  const times_worst_case = rankToAllSeconds(rank.worst_case,  profile);

  const bestAttackRank = rank.details[rank.best_attack]?.rank ?? null;
  let best_attack_seconds = null;
  if (bestAttackRank !== null) {
    const timesForBest = ALGORITHMS.map(a => rankToSeconds(bestAttackRank, a, profile));
    best_attack_seconds = Math.min(...timesForBest);
  }

  return {
    context,
    rank,
    times,
    times_optimistic,
    times_worst_case,
    best_attack: rank.best_attack,
    best_attack_seconds,
    profile,
  };
}

export function rankToAllProfiles(rank, algo) {
  return {
    amateur:      rankToSeconds(rank, algo, 'amateur'),
    experienced:  rankToSeconds(rank, algo, 'experienced'),
    professional: rankToSeconds(rank, algo, 'professional'),
    nation_state: rankToSeconds(rank, algo, 'nation_state'),
  };
}
```

```js
async function sha1(text) {
  const buffer = await crypto.subtle.digest('SHA-1', new TextEncoder().encode(text));
  return Array.from(new Uint8Array(buffer))
    .map(b => b.toString(16).padStart(2, '0').toUpperCase())
    .join('');
}

export async function checkHibp(password) {
  const hash   = await sha1(password);
  const prefix = hash.slice(0, 5);
  const suffix = hash.slice(5);

  const res  = await fetch(`https://api.pwnedpasswords.com/range/${prefix}`);
  const text = await res.text();

  const line = text.split('\r\n').find(l => l.startsWith(suffix));
  const count = line ? parseInt(line.split(':')[1], 10) : 0;

  return { hash, prefix, count, pwned: count > 0 };
}
```

```js
const DB_NAME    = 'time2crack-v2';
const DB_VERSION = 1;
const STORE      = 'cache';

function openDb() {
  return new Promise((resolve, reject) => {
    const req = indexedDB.open(DB_NAME, DB_VERSION);
    req.onupgradeneeded = e => e.target.result.createObjectStore(STORE);
    req.onsuccess = e => resolve(e.target.result);
    req.onerror   = e => reject(e.target.error);
  });
}

export async function getFromCache(key) {
  try {
    const db  = await openDb();
    const tx  = db.transaction(STORE, 'readonly');
    const req = tx.objectStore(STORE).get(key);
    return await new Promise((resolve, reject) => {
      req.onsuccess = () => resolve(req.result);
      req.onerror   = () => reject(req.error);
    });
  } catch {
    return undefined;
  }
}

export async function setInCache(key, value) {
  try {
    const db  = await openDb();
    const tx  = db.transaction(STORE, 'readwrite');
    tx.objectStore(STORE).put(value, key);
    return new Promise((resolve, reject) => {
      tx.oncomplete = resolve;
      tx.onerror    = () => reject(tx.error);
    });
  } catch {
    // private browsing — silently fail
  }
}
```

```js
import { qs } from '../utils/dom.js';
import { formatTime } from '../utils/format.js';

export function initResults() {
  // Results section starts in is-empty state (rendered by Nunjucks).

export function renderResults(crackResult) {
  const section = qs('#results');
  const tbody   = qs('#results-body');
  const table   = qs('.results__table');
  const emptyMsg = qs('.results__empty-msg');
  if (!section || !tbody || !table) return;

  section.classList.remove('is-empty');
  if (emptyMsg) emptyMsg.hidden = true;
  table.hidden = false;

  const attacks = Object.entries(crackResult.rank.details)
    .sort(([, a], [, b]) => a.rank - b.rank);

  tbody.innerHTML = attacks.map(([name, detail]) => {
    const timeStr = formatTime(
      crackResult.times[Object.keys(crackResult.times)[0]] *
      (detail.rank / crackResult.rank.standard)
    );
    const isBest = name === crackResult.best_attack;
    return `<tr${isBest ? ' class="results__row--best"' : ''}>
      <td>${name}${isBest ? ' ★' : ''}</td>
      <td>${timeStr}</td>
    </tr>`;
  }).join('');
}

export function resetResults() {
  const section  = qs('#results');
  const tbody    = qs('#results-body');
  const table    = qs('.results__table');
  const emptyMsg = qs('.results__empty-msg');
  if (!section) return;
  section.classList.add('is-empty');
  if (tbody)    tbody.innerHTML = '';
  if (table)    table.hidden = true;
  if (emptyMsg) emptyMsg.hidden = false;
}
```

```js
  on(toggle, 'click', () => {

  });
```
